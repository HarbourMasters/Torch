47class SpriteData : public IParsedData {
48 public:
49 int16_t mFrameCount; // animation frame count
50 int16_t mFormatCode; // texture format (RGBA16=0, RGBA32=1, CI4=2, CI8=3, etc.)
51 int16_t mUnk4; // ROM header field at offset 4
52 int16_t mUnk6; // ROM header field at offset 6
53 int16_t mUnk8; // display width, drives billboard vertex positioning
54 int16_t mUnkA; // display height, same
55 // unkC bitfield: the animation params, a BE u32 at ROM offset 0x0C
56 uint8_t mAnimSpeed; // bits 31-28: 4 bits — animation speed divisor
57 uint8_t mAnimType; // bits 27-25: 3 bits — animation type (0=none, 1-4=various
58 // loop modes)
59 uint8_t mAnimDirection; // bits 24-23: 2 bits — animation direction control
60 uint8_t mAnimFlip; // bits 22-21: 2 bits — flip/mirror control
61 std::vector<uint16_t> mChunkCounts; // chunks per frame (length = mFrameCount)
62 std::vector<std::pair<int16_t, int16_t>> mPositions; // (x, y) offset per chunk
63 std::vector<SpriteFrameHeader> mFrameHeaders; // per-frame header data
64 std::vector<SpriteFrame> mFrames; // empty for a sprite read without its texels
65 // Built from modding files, so the binary export writes the chunk textures too
66 bool mWriteTextures = false;
67
68 SpriteData(int16_t frameCount, int16_t formatCode, std::vector<uint16_t> chunkCounts,
69 std::vector<std::pair<int16_t, int16_t>> positions, std::vector<SpriteFrameHeader> frameHeaders = {},
70 int16_t unk4 = 0, int16_t unk6 = 0, int16_t unk8 = 0, int16_t unkA = 0, uint8_t animSpeed = 0,
71 uint8_t animType = 0, uint8_t animDirection = 0, uint8_t animFlip = 0)
72 : mFrameCount(frameCount), mFormatCode(formatCode), mUnk4(unk4), mUnk6(unk6), mUnk8(unk8), mUnkA(unkA),
73 mAnimSpeed(animSpeed), mAnimType(animType), mAnimDirection(animDirection), mAnimFlip(animFlip),
74 mChunkCounts(std::move(chunkCounts)), mPositions(std::move(positions)),
75 mFrameHeaders(std::move(frameHeaders)) {
76 }
77};
