182// Takes SM_SPIRAL_MOUNTAIN or MAP_1_SM_SPIRAL_MOUNTAIN.
183inline std::optional<uint16_t> MapFromName(const std::string& name) {
184 for (uint16_t map = 1; map < sizeof(kMapNames) / sizeof(kMapNames[0]); map++) {
185 const std::string shortName = kMapNames[map];
186 if (name == shortName ||
187 (name.rfind("MAP_", 0) == 0 && name.size() > shortName.size() &&
188 name.compare(name.size() - shortName.size() - 1, std::string::npos, "_" + shortName) == 0)) {
189 return map;
190 }
191 }
192 return std::nullopt;
193}
