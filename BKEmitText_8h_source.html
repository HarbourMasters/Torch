18inline void EmitText(YAML::Emitter& out, const std::string& str) {
19 const char* s = str.c_str();
20 bool plain = true;
21 for (const char* p = s; *p != '\0'; p++) {
22 if (static_cast<uint8_t>(*p) < 0x20 || static_cast<uint8_t>(*p) > 0x7E || *p == '\\') {
23 plain = false;
24 break;
25 }
26 }
27 if (plain) {
28 out << YAML::DoubleQuoted << s;
29 return;
30 }
31 std::string escaped;
32 for (const char* p = s; *p != '\0'; p++) {
33 const auto c = static_cast<uint8_t>(*p);
34 if (c == '\\') {
35 escaped += "\\\\";
36 } else if (c < 0x20 || c > 0x7E) {
37 char buf[5];
38 std::snprintf(buf, sizeof(buf), "\\x%02X", c);
39 escaped += buf;
40 } else {
41 escaped += static_cast<char>(c);
42 }
43 }
44 out << YAML::SingleQuoted << escaped;
45}

54 out += '\\';

57 }
